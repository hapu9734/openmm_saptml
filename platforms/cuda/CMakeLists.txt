# CUDA platform of the SAPT-ML plugin.

SET(SHARED_TARGET ${SAPTML_CUDA_LIBRARY_NAME})

FILE(GLOB SOURCE_FILES ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp ${CMAKE_SOURCE_DIR}/platforms/common/src/*.cpp)
FILE(GLOB SOURCE_INCLUDE_FILES ${CMAKE_CURRENT_SOURCE_DIR}/src/*.h ${CMAKE_SOURCE_DIR}/platforms/common/src/*.h)
FILE(GLOB API_INCLUDE_FILES ${CMAKE_CURRENT_SOURCE_DIR}/include/*.h)
SET(COMMON_KERNELS_CPP ${CMAKE_BINARY_DIR}/platforms/common/src/CommonSAPTMLKernelSources.cpp)
LIST(APPEND SOURCE_FILES ${COMMON_KERNELS_CPP})
SET_SOURCE_FILES_PROPERTIES(${COMMON_KERNELS_CPP} PROPERTIES GENERATED TRUE)

INCLUDE_DIRECTORIES(BEFORE ${CMAKE_CURRENT_SOURCE_DIR}/include)
INCLUDE_DIRECTORIES(BEFORE ${CMAKE_CURRENT_SOURCE_DIR}/src)
INCLUDE_DIRECTORIES(BEFORE ${CMAKE_SOURCE_DIR}/platforms/common/src)
INCLUDE_DIRECTORIES(BEFORE ${CMAKE_BINARY_DIR}/platforms/common/src)
INCLUDE_DIRECTORIES(${CUDAToolkit_INCLUDE_DIRS})

ADD_LIBRARY(${SHARED_TARGET} SHARED ${SOURCE_FILES} ${SOURCE_INCLUDE_FILES} ${API_INCLUDE_FILES})
ADD_DEPENDENCIES(${SHARED_TARGET} SAPTMLCommonKernels)
# OpenMM 8.3 and later provide a platform FFT (ComputeContext::createFFT); older versions use cuFFT directly.
IF(EXISTS "${OPENMM_DIR}/include/openmm/common/FFT3D.h")
    TARGET_COMPILE_DEFINITIONS(${SHARED_TARGET} PRIVATE SAPTML_USE_COMPUTE_FFT)
    TARGET_LINK_LIBRARIES(${SHARED_TARGET} ${SHARED_SAPTML_TARGET} ${OPENMM_LIBRARY} ${OPENMM_CUDA_LIBRARY} CUDA::cuda_driver)
ELSE()
    TARGET_LINK_LIBRARIES(${SHARED_TARGET} ${SHARED_SAPTML_TARGET} ${OPENMM_LIBRARY} ${OPENMM_CUDA_LIBRARY} CUDA::cuda_driver CUDA::cufft)
ENDIF()
SET_TARGET_PROPERTIES(${SHARED_TARGET} PROPERTIES
    COMPILE_FLAGS "-DOPENMM_BUILDING_SHARED_LIBRARY ${EXTRA_COMPILE_FLAGS}"
    LINK_FLAGS "${EXTRA_COMPILE_FLAGS}")
INSTALL(TARGETS ${SHARED_TARGET} DESTINATION lib/plugins)

SUBDIRS(tests)
